// DAY 163: ADD A SUB-DOCUMENT ROUTE + MONGODB INDEXES

// Install dependencies:
// npm install express mongoose dotenv

// Create a .env file:
// MONGO_URI=mongodb://127.0.0.1:27017/day163db
// PORT=3000

const express = require("express");
const mongoose = require("mongoose");
require("dotenv").config();

const app = express();

app.use(express.json());

// =====================================
// 1. REVIEW SUB-DOCUMENT SCHEMA
// =====================================

const reviewSchema = new mongoose.Schema(
  {
    reviewer: {
      type: String,
      required: true,
      trim: true,
    },

    rating: {
      type: Number,
      required: true,
      min: 1,
      max: 5,
    },

    comment: {
      type: String,
      trim: true,
    },
  },
  { timestamps: true }
);

// =====================================
// 2. PRODUCT SCHEMA
// =====================================

const productSchema = new mongoose.Schema(
  {
    name: {
      type: String,
      required: true,
      trim: true,
    },

    category: {
      type: String,
      required: true,
    },

    price: {
      type: Number,
      required: true,
    },

    email: {
      type: String,
      unique: true,
      sparse: true,
    },

    reviews: [reviewSchema],
  },
  { timestamps: true }
);

// =====================================
// 3. MONGODB INDEXES
// =====================================

// Single-field index
productSchema.index({ name: 1 });

// Compound index
productSchema.index({ category: 1, price: -1 });

// Unique index on email is created by the unique option above.

// =====================================
// 4. PRODUCT MODEL
// =====================================

const Product = mongoose.model("Product", productSchema);

// =====================================
// 5. CREATE A PRODUCT
// =====================================

app.post("/api/products", async (req, res) => {
  try {
    const product = await Product.create(req.body);

    res.status(201).json({
      message: "Product created successfully",
      product,
    });
  } catch (error) {
    res.status(400).json({
      message: error.message,
    });
  }
});

// =====================================
// 6. GET ALL PRODUCTS
// =====================================

app.get("/api/products", async (req, res) => {
  try {
    const products = await Product.find();

    res.status(200).json(products);
  } catch (error) {
    res.status(500).json({
      message: error.message,
    });
  }
});

// =====================================
// 7. ADD A REVIEW (DAY 163 MAIN TASK)
// =====================================

app.post("/api/products/:id/reviews", async (req, res) => {
  try {
    const product = await Product.findById(req.params.id);

    if (!product) {
      return res.status(404).json({
        message: "Product not found",
      });
    }

    product.reviews.push({
      reviewer: req.body.reviewer,
      rating: req.body.rating,
      comment: req.body.comment,
    });

    await product.save();

    res.status(201).json({
      message: "Review added successfully",
      product,
    });
  } catch (error) {
    res.status(400).json({
      message: error.message,
    });
  }
});

// =====================================
// 8. GET ALL REVIEWS FOR A PRODUCT
// =====================================

app.get("/api/products/:id/reviews", async (req, res) => {
  try {
    const product = await Product.findById(req.params.id);

    if (!product) {
      return res.status(404).json({
        message: "Product not found",
      });
    }

    res.status(200).json(product.reviews);
  } catch (error) {
    res.status(400).json({
      message: error.message,
    });
  }
});

// =====================================
// 9. UPDATE A REVIEW
// =====================================

app.put("/api/products/:id/reviews/:reviewId", async (req, res) => {
  try {
    const product = await Product.findById(req.params.id);

    if (!product) {
      return res.status(404).json({
        message: "Product not found",
      });
    }

    const review = product.reviews.id(req.params.reviewId);

    if (!review) {
      return res.status(404).json({
        message: "Review not found",
      });
    }

    if (req.body.reviewer !== undefined) {
      review.reviewer = req.body.reviewer;
    }

    if (req.body.rating !== undefined) {
      review.rating = req.body.rating;
    }

    if (req.body.comment !== undefined) {
      review.comment = req.body.comment;
    }

    await product.save();

    res.status(200).json({
      message: "Review updated successfully",
      review,
    });
  } catch (error) {
    res.status(400).json({
      message: error.message,
    });
  }
});

// =====================================
// 10. DELETE A REVIEW
// =====================================

app.delete("/api/products/:id/reviews/:reviewId", async (req, res) => {
  try {
    const product = await Product.findById(req.params.id);

    if (!product) {
      return res.status(404).json({
        message: "Product not found",
      });
    }

    const review = product.reviews.id(req.params.reviewId);

    if (!review) {
      return res.status(404).json({
        message: "Review not found",
      });
    }

    review.deleteOne();

    await product.save();

    res.status(200).json({
      message: "Review deleted successfully",
    });
  } catch (error) {
    res.status(400).json({
      message: error.message,
    });
  }
});

// =====================================
// 11. START SERVER
// =====================================

const PORT = process.env.PORT || 3000;

async function startServer() {
  try {
    await mongoose.connect(process.env.MONGO_URI);

    console.log("MongoDB connected successfully");

    app.listen(PORT, () => {
      console.log(`Server running at http://localhost:${PORT}`);
    });
  } catch (error) {
    console.error("Database connection failed:", error.message);
    process.exit(1);
  }
}

startServer();
