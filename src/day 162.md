require("dotenv").config();

const express = require("express");
const mongoose = require("mongoose");

const app = express();

app.use(express.json());


// ==============================================
// 1. REVIEW SUB-DOCUMENT SCHEMA
// ==============================================

const reviewSchema = new mongoose.Schema({
    reviewer: {
        type: String,
        required: true
    },

    rating: {
        type: Number,
        required: true,
        min: 1,
        max: 5
    },

    comment: {
        type: String,
        default: ""
    }
});


// ==============================================
// 2. PRODUCT SCHEMA
// ==============================================

const productSchema = new mongoose.Schema({

    name: {
        type: String,
        required: true
    },

    category: {
        type: String,
        required: true
    },

    price: {
        type: Number,
        required: true
    },

    stock: {
        type: Number,
        default: 0
    },

    // Embedded sub-documents
    reviews: [reviewSchema]

}, { timestamps: true });


const Product = mongoose.model("Product", productSchema);


// ==============================================
// 3. CREATE PRODUCT WITH EMBEDDED REVIEWS
// ==============================================

app.post("/api/products", async (req, res) => {
    try {

        const product = await Product.create(req.body);

        res.status(201).json({
            message: "Product created successfully",
            product
        });

    } catch (error) {

        res.status(400).json({
            error: error.message
        });

    }
});


// ==============================================
// 4. GET ALL PRODUCTS
// ==============================================

app.get("/api/products", async (req, res) => {
    try {

        const products = await Product.find();

        res.json(products);

    } catch (error) {

        res.status(500).json({
            error: error.message
        });

    }
});


// ==============================================
// 5. QUERY OPERATORS
// ==============================================

// $gt - Greater than

app.get("/api/products/greater/:price", async (req, res) => {
    try {

        const products = await Product.find({
            price: { $gt: Number(req.params.price) }
        });

        res.json(products);

    } catch (error) {

        res.status(500).json({ error: error.message });

    }
});


// $gte - Greater than or equal

app.get("/api/products/minimum/:price", async (req, res) => {
    try {

        const products = await Product.find({
            price: { $gte: Number(req.params.price) }
        });

        res.json(products);

    } catch (error) {

        res.status(500).json({ error: error.message });

    }
});


// $lt - Less than

app.get("/api/products/less/:price", async (req, res) => {
    try {

        const products = await Product.find({
            price: { $lt: Number(req.params.price) }
        });

        res.json(products);

    } catch (error) {

        res.status(500).json({ error: error.message });

    }
});


// $lte - Less than or equal

app.get("/api/products/maximum/:price", async (req, res) => {
    try {

        const products = await Product.find({
            price: { $lte: Number(req.params.price) }
        });

        res.json(products);

    } catch (error) {

        res.status(500).json({ error: error.message });

    }
});


// $in - Match any value in an array

app.get("/api/products/categories", async (req, res) => {
    try {

        const products = await Product.find({
            category: {
                $in: ["Electronics", "Clothing"]
            }
        });

        res.json(products);

    } catch (error) {

        res.status(500).json({ error: error.message });

    }
});


// $nin - Exclude values in an array

app.get("/api/products/exclude", async (req, res) => {
    try {

        const products = await Product.find({
            category: {
                $nin: ["Electronics", "Clothing"]
            }
        });

        res.json(products);

    } catch (error) {

        res.status(500).json({ error: error.message });

    }
});


// $ne - Not equal

app.get("/api/products/available", async (req, res) => {
    try {

        const products = await Product.find({
            stock: { $ne: 0 }
        });

        res.json(products);

    } catch (error) {

        res.status(500).json({ error: error.message });

    }
});


// $or - Either condition

app.get("/api/products/or", async (req, res) => {
    try {

        const products = await Product.find({
            $or: [
                { price: { $lt: 100 } },
                { stock: { $gt: 50 } }
            ]
        });

        res.json(products);

    } catch (error) {

        res.status(500).json({ error: error.message });

    }
});


// $and - Both conditions

app.get("/api/products/and", async (req, res) => {
    try {

        const products = await Product.find({
            $and: [
                { price: { $gt: 50 } },
                { stock: { $gt: 0 } }
            ]
        });

        res.json(products);

    } catch (error) {

        res.status(500).json({ error: error.message });

    }
});


// ==============================================
// 6. ADD EMBEDDED REVIEW
// ==============================================

app.post("/api/products/:id/reviews", async (req, res) => {
    try {

        const product = await Product.findById(req.params.id);

        if (!product) {
            return res.status(404).json({
                message: "Product not found"
            });
        }

        product.reviews.push(req.body);

        await product.save();

        res.status(201).json({
            message: "Review added successfully",
            reviews: product.reviews
        });

    } catch (error) {

        res.status(400).json({
            error: error.message
        });

    }
});


// ==============================================
// 7. GET EMBEDDED REVIEWS
// ==============================================

app.get("/api/products/:id/reviews", async (req, res) => {
    try {

        const product = await Product.findById(req.params.id);

        if (!product) {
            return res.status(404).json({
                message: "Product not found"
            });
        }

        res.json({
            product: product.name,
            reviews: product.reviews
        });

    } catch (error) {

        res.status(500).json({
            error: error.message
        });

    }
});


// ==============================================
// 8. UPDATE EMBEDDED REVIEW
// ==============================================

app.patch(
    "/api/products/:productId/reviews/:reviewId",
    async (req, res) => {
        try {

            const product = await Product.findById(
                req.params.productId
            );

            if (!product) {
                return res.status(404).json({
                    message: "Product not found"
                });
            }

            const review = product.reviews.id(
                req.params.reviewId
            );

            if (!review) {
                return res.status(404).json({
                    message: "Review not found"
                });
            }

            if (req.body.rating !== undefined) {
                review.rating = req.body.rating;
            }

            if (req.body.comment !== undefined) {
                review.comment = req.body.comment;
            }

            await product.save();

            res.json({
                message: "Review updated successfully",
                review
            });

        } catch (error) {

            res.status(400).json({
                error: error.message
            });

        }
    }
);


// ==============================================
// 9. DELETE EMBEDDED REVIEW
// ==============================================

app.delete(
    "/api/products/:productId/reviews/:reviewId",
    async (req, res) => {
        try {

            const product = await Product.findById(
                req.params.productId
            );

            if (!product) {
                return res.status(404).json({
                    message: "Product not found"
                });
            }

            const review = product.reviews.id(
                req.params.reviewId
            );

            if (!review) {
                return res.status(404).json({
                    message: "Review not found"
                });
            }

            review.deleteOne();

            await product.save();

            res.json({
                message: "Review deleted successfully"
            });

        } catch (error) {

            res.status(500).json({
                error: error.message
            });

        }
    }
);


// ==============================================
// 10. UPDATE PRODUCT
// ==============================================

app.patch("/api/products/:id", async (req, res) => {
    try {

        const product = await Product.findByIdAndUpdate(
            req.params.id,
            req.body,
            {
                new: true,
                runValidators: true
            }
        );

        if (!product) {
            return res.status(404).json({
                message: "Product not found"
            });
        }

        res.json({
            message: "Product updated successfully",
            product
        });

    } catch (error) {

        res.status(400).json({
            error: error.message
        });

    }
});


// ==============================================
// 11. DELETE PRODUCT
// ==============================================

app.delete("/api/products/:id", async (req, res) => {
    try {

        const product = await Product.findByIdAndDelete(
            req.params.id
        );

        if (!product) {
            return res.status(404).json({
                message: "Product not found"
            });
        }

        res.json({
            message: "Product deleted successfully"
        });

    } catch (error) {

        res.status(500).json({
            error: error.message
        });

    }
});


// ==============================================
// 12. CONNECT DATABASE AND START SERVER
// ==============================================

const PORT = process.env.PORT || 5000;

mongoose.connect(process.env.MONGO_URI)
    .then(() => {

        console.log("MongoDB connected successfully");

        app.listen(PORT, () => {
            console.log(`Server running on port ${PORT}`);
        });

    })
    .catch(error => {

        console.error("MongoDB connection failed:", error);

    });
